Marginal objective
& $\begin{gathered}
\mathcal{L}_{\mathrm{FM}}(\theta)=\\
\mathbb{E}_{t,p_t(x)}\!\left[
\left\lVert v_t(x)-u_t(x)\right\rVert^2\right]\\
\text{(Eq.~5)}
\end{gathered}$
& $\begin{gathered}
\mathcal{L}_{\mathrm{RFM}}(\theta)=\\
\mathbb{E}_{t,p_t(x)}\!\left[
\left\lVert v_t(x)-u_t(x)\right\rVert_g^2\right]\\
\text{(Eq.~3)}
\end{gathered}$ \\
\hline
Conditional objective
& $\begin{gathered}
\mathcal{L}_{\mathrm{CFM}}(\theta)=\\
\mathbb{E}_{\substack{t,q(x_1),\\p_t(x\mid x_1)}}\!\Bigl[\\
\left\lVert v_t(x)-u_t(x\mid x_1)\right\rVert^2\Bigr]\\
\text{(Eq.~9)}
\end{gathered}$
& $\begin{gathered}
\mathcal{L}_{\mathrm{RCFM}}(\theta)=\\
\mathbb{E}_{\substack{t,q(x_1),\\p_t(x\mid x_1)}}\!\Bigl[\\
\left\lVert v_t(x)-u_t(x\mid x_1)\right\rVert_g^2\Bigr]\\
\text{(Eq.~8)}
\end{gathered}$ \\
\hline
Conditional velocity $u_t$
& $\begin{gathered}
u_t(x\mid x_1)=\\
\dfrac{x_1-(1-\sigma_{\min})x}
{1-(1-\sigma_{\min})t}\\
\text{(Eq.~21)}
\end{gathered}$
& $\begin{gathered}
u_t(x\mid x_1)=\\
\dfrac{d\log\kappa(t)}{dt}\,d(x,x_1)\\
{}\times\dfrac{\nabla d(x,x_1)}
{\left\lVert\nabla d(x,x_1)\right\rVert_g^2}\\
\text{(Eq.~13)}
\end{gathered}$ \\
\hline
Conditional objective explicit form
& $\begin{gathered}
\mathcal{L}_{\mathrm{CFM}}(\theta)=\\
\mathbb{E}_{\substack{t,q(x_1),\\p(x_0)}}\!\left[
\left\lVert v_t\!\left(\psi_t(x_0)\right)\right.\right.\\
\left.\left.{}-\left(x_1-(1-\sigma_{\min})x_0\right)
\right\rVert^2\right]\\
\text{(Eq.~23)}
\end{gathered}$
& $\begin{gathered}
\mathcal{L}_{\mathrm{RCFM}}(\theta)=\\
\mathbb{E}_{\substack{t,q(x_1),\\p(x_0)}}\!\left[
\left\lVert v_t(x_t)+d(x_0,x_1)\right.\right.\\
\left.\left.{}\times\dfrac{\nabla d(x_t,x_1)}
{\left\lVert\nabla d(x_t,x_1)\right\rVert_g^2}
\right\rVert_g^2\right]\\
\text{(Eq.~14)}
\end{gathered}$ \\
\hline
$x_t$ definition
& $\begin{gathered}
x_t=\psi_t(x_0)=\\
\left[1-(1-\sigma_{\min})t\right]x_0+t x_1\\
\text{(Eq.~22)}
\end{gathered}$
& $\begin{gathered}
x_t=\exp_{x_1}\!\left(
\kappa(t)\log_{x_1}(x_0)\right),\\
t\in[0,1]\\
\text{(Eq.~15)}
\end{gathered}$ \\
